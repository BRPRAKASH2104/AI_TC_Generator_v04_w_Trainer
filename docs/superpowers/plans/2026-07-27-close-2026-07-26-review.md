  ```python
  def _match_pairs(
      self,
      generated_cases: list[dict[str, Any]],
      reference_cases: list[dict[str, Any]],
  ) -> list[tuple[int, int]]:
      """Return a deterministic maximum-cardinality bipartite matching.

      Args:
          generated_cases: Canonical generated test cases.
          reference_cases: Canonical reference test cases.

      Returns:
          Unique ``(generated_index, reference_index)`` pairs.
      """
      adjacency = [
          [
              reference_index
              for reference_index, reference_case in enumerate(reference_cases)
              if self._dedup.similarity(generated_case, reference_case)
              >= self.match_threshold
          ]
          for generated_case in generated_cases
      ]
      reference_to_generated: dict[int, int] = {}

      def assign(generated_index: int, seen: set[int]) -> bool:
          for reference_index in adjacency[generated_index]:
              if reference_index in seen:
                  continue
              seen.add(reference_index)
              previous_generated = reference_to_generated.get(reference_index)
              if previous_generated is None or assign(previous_generated, seen):
                  reference_to_generated[reference_index] = generated_index
                  return True
          return False

      for generated_index in range(len(generated_cases)):
          assign(generated_index, set())

      return sorted(
          (generated_index, reference_index)
          for reference_index, generated_index in reference_to_generated.items()
      )
  ```
